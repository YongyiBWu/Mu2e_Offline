//
//     W_width + 2*leak + 2*side_thickness = top shield width

//
// This is NOT the front/back clearance; see ZGapBack = 4.35 mm.
double stm.STM_SSC.leak      =  10.000;

// Bore radii, as updated imperial sizes. The two bores are stepped

// The SSC sits inside the steel cradle, which is deeper than it is:
//
//     cradle depth (SteelSmallSSC)  110.3 mm
//     SSC z-extent                  101.6 mm  (Wdepth_f + Wdepth_b)
//     ZGap = 110.3 - 101.6 =          8.7 mm  total slack
//
// ZGapBack is the BACK clearance and ZGap-ZGapBack the front, so the
// SSC block centre lands at Wdepth_f + ZGap - ZGapBack = 55.15 mm and
// the block spans z = [4.35, 105.95] inside the cradle's [0, 110.3].
// As extracted the SSC is centred in the cradle, hence ZGapBack =
// ZGap/2; it stays independently tunable.
//
// Note what the code actually places: the SSC is a union of two slabs
// and a G4UnionSolid takes its origin from the FIRST constituent, so
// the placement point is the FRONT slab's centre,
//     Wdepth_f/2 + ZGap - ZGapBack = 29.75 mm
// not the block centre above.
double stm.STM_SSC.ZGap            = 8.70;
double stm.STM_SSC.ZGapBack        = 4.35;

// Height of the SSC bore axis above the baseplate. Five independent
// stacks through the shielding agree on this to within 0.5 mm; 236.000
// is the value two all-imperial paths give exactly.
double stm.STM_SSC.boreToBase      = 236.000;


// ---- Steel support cradle ----------------------------------------
//
// Five plates: two sides, a base under the SSC, a top over it, and a
// bottom plate resting on the baseplate. Its front face is
// STMShieldingRef, the origin every other piece is placed against.
//
// Nothing here is positioned independently. The plates build outward
// from the SSC block, so every width and offset is derived and the
// relations close against the CAD to within 0.06 mm:
//
//     base width  = W_width + 2*leak             = 182.560  (CAD 182.574)
//     top width   = W_width + 2*leak + 2*side_T  = 220.660  (CAD 220.672)
//     bottom      = same as top                  = 220.660  (CAD 220.660)
//
//     side  dx = +/-(W_width/2  + leak + side_T/2) = +/-100.805
//     top   dy =   +(W_height/2 + leak + top_T/2)  =  +60.165
//     base  dy =   -(W_height/2 + leak + base_T/2) =  -60.165
//
// so only the thicknesses and the side-plate height are free. All
// five plates are 110.3 mm deep, and that depth is what sets ZGap
// against the SSC's 101.6 mm.
//
// The side plates hang from the top of the cradle: their top edge is
// level with the base plate's underside at W_height/2 + leak above the
// bore, and side_H then carries the stack down to the bottom plate,
// which rests on the baseplate. That chain is what fixes side_H:
//
//     side_H - (W_height/2 + leak) + bottom_T = 236.000
//     273.940 -        50.640      +  12.700  = 236.000
//
// i.e. the bore sits 236 mm above the baseplate. CAD measures the side
// plate at 273.187, but that leaves a 0.75 mm crack; 273.940 is the
// height that closes the stack, and is the value the other four paths
// to the baseplate agree with.
bool   stm.SSCSupport.build       = true;

// The CAD part labels give these plates as mild steel.
string stm.SSCSupport.material    = "MildSteel";
double stm.SSCSupport.depth       = 110.300;  // 4.343 in, sets ZGap
double stm.SSCSupport.side_T      =  19.050;  // 0.750 in, side plate thickness
double stm.SSCSupport.side_H      = 273.940;  // closes the stack; CAD 273.187
double stm.SSCSupport.base_T      =  19.050;  // 0.750 in
double stm.SSCSupport.bottom_T    =  12.700;  // 0.500 in
double stm.SSCSupport.top_T       =  19.050;  // 0.750 in

//      block abuts the right half's -x course edge.
//   y  the brick block stands on the baseplate and rises 16 in.
//   z  the second layer (lead brick) shares the plane of the right 
//      half's last lead layer; the poly sheets and the bricks behind 
//      them stack on downstream of it.
bool   stm.FrontShieldingLeft.build = true;

// All the poly here is the same borated grade as the right half's.
string stm.FrontShieldingLeft.BPmaterial = "Polyethylene092B050d095";

// Each sheet gives its CENTRE as fromEndX, measured from
// stm.FrontShieldingRight.brickEndX -- the same reference the brick
// block is flush with. The offsets run -x into this half and so are
// negative. The sheets are not all oriented alike, so dx is a
// thickness on some and a width on others; each says which below.

// The standalone slab against the right half's -x face, spanning the
// depth of its first four layers. This one stands on edge: its 1 in is
// the thickness, in x, and its 490 mm is the same as the total height
// of front shielding right. It hangs off the far side of the 24 in of
// courses, so its centre is that span plus half that thickness.
double stm.FrontShieldingLeft.sidePoly.fromEndX = -622.300;  // -(24 + 0.5) in
double stm.FrontShieldingLeft.sidePoly.dx =  25.400;  // 1 in, THICKNESS
double stm.FrontShieldingLeft.sidePoly.dy = 490.000;
double stm.FrontShieldingLeft.sidePoly.dz = 152.200;  // 6 in

// The outer sheet, one layer ahead of the brick block. This one lies
// in the xy plane, so its 1 in is in z and dx is a width. Its +x face
// stands 25 in back from the reference and the centre follows.
double stm.FrontShieldingLeft.outerPoly.fromEndX = -875.600;  // -25 in - dx/2
double stm.FrontShieldingLeft.outerPoly.dx = 481.200;  // as measured, WIDTH
double stm.FrontShieldingLeft.outerPoly.dy = 406.200;  // 16 in
double stm.FrontShieldingLeft.outerPoly.dz =  25.400;  //  1 in, thickness

// The brick block: three columns in x, each 16 in tall, sharing the
// plane of the right half's last lead layer. The outer two are stacks
// of four 4 in courses; the middle is a single 16 in brick on end, so
// the pitch is mixed and the columns are given separately. Widths run
// 8 in, 4 in, 8 in for 20 in overall, the +x face flush with the
// right half and the bottom on the baseplate.
//
// A column is a list of stm.leadBrick types read from -y to +y, butted
// end to end and flush on the baseplate, in the same form the right
// half writes its courses.
//
// Orientation is per column rather than shared, because the columns do
// not agree: the outer two lie in the leadBrick native frame, 8 in
// along x and 4 in along y, while the middle brick is stood on end so
// its 16 in runs up y.
int stm.FrontShieldingLeft.blockColumnN = 3;
vector<int> stm.FrontShieldingLeft.blockColumn1 = {2,2,2,2};  // four 2x4x8 stacked
vector<int> stm.FrontShieldingLeft.blockColumn2 = {1};        // one 2x4x16 on end
vector<int> stm.FrontShieldingLeft.blockColumn3 = {2,2,2,2};  // four 2x4x8 stacked

string stm.FrontShieldingLeft.blockColumn1Orientation = "000";
string stm.FrontShieldingLeft.blockColumn2Orientation = "001";
string stm.FrontShieldingLeft.blockColumn3Orientation = "000";

// The inner sheet and the edge sheet beside it. Both are 16 in wide;
// they are given apart because they sit at quite different heights, so
// each carries its own bottomToBase.
//
// The inner sheet reaches the floor like the rest of this half.
double stm.FrontShieldingLeft.innerPoly.fromEndX     = -736.600;  // -21-8=-29 in
double stm.FrontShieldingLeft.innerPoly.bottomToBase =    0.000;
double stm.FrontShieldingLeft.innerPoly.dx = 406.400;  // 16 in
double stm.FrontShieldingLeft.innerPoly.dy = 302.800;  // as measured, up to 304.8 mm / 12 in
double stm.FrontShieldingLeft.innerPoly.dz =  25.400;  //  1 in

// The edge sheet does not: it rides high on the aluminum top wall,
// clear of the inner sheet below it.
double stm.FrontShieldingLeft.edgePoly.fromEndX     = -787.400;  // -23-8=-31 in
double stm.FrontShieldingLeft.edgePoly.bottomToBase =  323.850;  // 12.75 in
double stm.FrontShieldingLeft.edgePoly.dx = 406.400;  // 16 in
double stm.FrontShieldingLeft.edgePoly.dy = 101.600;  //  4 in
double stm.FrontShieldingLeft.edgePoly.dz =  25.400;  //  1 in

// The two bricks beside the edge sheet: one course of stm.leadBrick
// types read from -x to +x, in the same form the block columns and the
// right half's courses use. They are level with the edge sheet rather
// than stacked on it, so they take its bottomToBase; STMMaker works
// out where the course starts.
vector<int> stm.FrontShieldingLeft.edgeCourse = {2,2};
string stm.FrontShieldingLeft.edgeCourseOrientation = "000";

// Behind the inner sheet, a grid sharing its x extent: three courses
// of two 8 in bricks, read -x to +x and stacked -y to +y on a 4 in

int stm.FrontShieldingLeft.innerGridCourseN = 3;
vector<int> stm.FrontShieldingLeft.innerGridCourse1 = {2,2};
vector<int> stm.FrontShieldingLeft.innerGridCourse2 = {2,2};
vector<int> stm.FrontShieldingLeft.innerGridCourse3 = {2,2};
string stm.FrontShieldingLeft.innerGridOrientation = "000";

// The lead triangle: two perpendicular 3 in sides swept 6 in along y.
//
// Placement is by ANCHOR rather than by centre, the same way the right
// half's copper lining is placed. The outline is drawn so that under
// orientation "300" the prism's local origin is its RIGHT ANGLE, the
// corner the two legs meet at, and STMMaker puts that point at:
//
//   x  the inner sheet's +x face,  innerPoly.fromEndX + dx/2
//   z  the inner sheet's -z face, which the right angle butts against
//   y  its top is level with the inner sheet's top. The extrusion is
//      centred on its origin, so the anchor sits HALF a sweep length
//      below that top.
//
// Its 3 in of z then reaches across the inner sheet and the 2x3 grid
// of bricks together, and the 6 in sweep is centred on the anchor.
string stm.FrontShieldingLeft.triangle.material = "G4_Pb";
vector<double> stm.FrontShieldingLeft.triangle.UVerts = {0.0, 76.2,   0.0};
vector<double> stm.FrontShieldingLeft.triangle.VVerts = {0.0,  0.0, -76.2};
double stm.FrontShieldingLeft.triangle.length      = 152.400;  // 6 in
string stm.FrontShieldingLeft.triangle.orientation = "300";
